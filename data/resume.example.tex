% Base resume for please-hire-me. Jake's Resume, the most used software-engineering template on
% Overleaf: https://github.com/jakegut/resume (MIT, Jake Gutierrez, based on sb2nov/resume).
%
% Changes from the original, all for ATS parsing under tectonic (XeTeX):
%   * Fonts load through fontspec with Ligatures=NoCommon, so the text layer says "fi", "ff",
%     "fl" instead of ligature glyphs that ATS parsers drop or garble. Same Latin Modern look.
%   * pdfTeX-only \input{glyphtounicode} / \pdfgentounicode are guarded.
%
% Setup copies this to data/resume.tex and fills it from your resume, word for word. The agent
% never edits data/resume.tex during a run. Per job it copies it to applications/files/<slug>/,
% tailors the copy, and builds it with `python3 scripts/app_docs.py render-resume`.
%
% Write special characters escaped: \% \$ \& \# \_ . An unescaped % silently deletes the rest of
% the line; render-resume catches it, but only after the fact.
%-------------------------------------------------------------------------------------------------

% Commands. The agent relies on these names: one \resumeItem per bullet, one \resumeSubheading
% per job, one \resumeProjectHeading per project.
\newcommand{\resumeItem}[1]{\item\small{{#1 \vspace{-2pt}}}}

\newcommand{\resumeSubheading}[4]{
  \vspace{-2pt}\item
    \begin{tabular*}{0.97\textwidth}[t]{l@{\extracolsep{\fill}}r}
      \textbf{#1} & #2 \\
      \textit{\small#3} & \textit{\small #4} \\
    \end{tabular*}\vspace{-7pt}
}

\newcommand{\resumeProjectHeading}[2]{
    \item
    \begin{tabular*}{0.97\textwidth}{l@{\extracolsep{\fill}}r}
      \small#1 & #2 \\
    \end{tabular*}\vspace{-7pt}
}

\renewcommand\labelitemii{$\vcenter{\hbox{\tiny$\bullet$}}$}
\newcommand{\resumeSubHeadingListStart}{\begin{itemize}[leftmargin=0.15in, label={}]}
\newcommand{\resumeSubHeadingListEnd}{\end{itemize}}
\newcommand{\resumeItemListStart}{\begin{itemize}}
\newcommand{\resumeItemListEnd}{\end{itemize}\vspace{-5pt}}
